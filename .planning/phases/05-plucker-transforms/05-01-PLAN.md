```cpp
SpatialVector PluckerTransform::transformMotion(const SpatialVector &vec) const
{
    // X = [R, 0; -R[t]x, R]
    // return X * vec = [Rω, R(v - [t]xω)]
    Vector3d transformedAngular = rotation * vec.getAngular();
    Vector3d transformedLinear = rotation * (vec.getLinear() + skew(translation) * vec.getAngular());
    return SpatialVector(transformedAngular, transformedLinear);
}

SpatialVector PluckerTransform::transformForce(const SpatialVector &vec) const
{
    // X^{-T} * vec = [R(τ - [t]xf), Rf]
    Vector3d transformedAngular = rotation * (vec.getAngular() - skew(translation) * vec.getLinear());
    Vector3d transformedLinear = rotation * vec.getLinear();
    return SpatialVector(transformedAngular, transformedLinear);
}
```

```cpp
SpatialVector PluckerTransform::inverseTransformMotion(const SpatialVector &vec) const
{
    Vector3d transformedAngular = rotation.transpose() * vec.getAngular();
    Vector3d transformedLinear = rotation.transpose() * vec.getLinear() + skew(translation) * transformedAngular;
    return SpatialVector(transformedAngular, transformedLinear);
}

SpatialVector PluckerTransform::inverseTransformForce(const SpatialVector &vec) const
{
    Vector3d transformedLinear = rotation.transpose() * vec.getLinear();
    Vector3d transformedAngular = rotation.transpose() * vec.getAngular() + skew(translation) * transformedLinear;
    return SpatialVector(transformedAngular, transformedLinear);
}
```

    ```cpp
    Vector3d transformedAngular = rotation * vec.getAngular();
    Vector3d transformedLinear = rotation * (vec.getLinear() + skew(translation) * vec.getAngular());
    ```
    
    **Featherstone (2008) Eq 2.43:**
    For motion transform X*v where X = [R, 0; -R*r̂, R]:
    - ω' = R*ω
    - v' = R*(v + r × ω) = R*(v - r̂*ω)
    
    **Verification steps:**
    1. Compare: code uses `vec.getLinear() + skew(translation) * vec.getAngular()`
    2. Featherstone: v' = R*(v + r × ω) where r × ω = -r̂*ω
    3. skew(r)*ω = r × ω, so formula should be: v + skew(translation) × ω
    4. Current code: v + skew(translation)*ω ✓ CORRECT
    
    **If formula matches:** No changes needed
    **If formula differs:** Fix to match Featherstone Eq 2.43

    
    **Featherstone (2008) Eq 2.44:**
    For force transform X^{-T}*f where X^{-T} = [R, R*r̂; 0, R]:
    - τ' = R*(τ + r × f) = R*(τ + r̂*f)
    - f' = R*f
    
    **Verification steps:**

    2. Featherstone: τ' = R*(τ + r × f) = R*(τ + r̂*f)
    3. SIGN MISMATCH: code uses minus, Featherstone uses plus
    4. Note: r̂*f = r × f, and skew(r)*f = r × f
    5. **BUG FOUND:** Should be `+ skew(translation) * vec.getLinear()` not minus
    
    **Fix if needed:**
    Change line 37 from:
    `vec.getAngular() - skew(translation) * vec.getLinear()`
    To:
    `vec.getAngular() + skew(translation) * vec.getLinear()`
