```cpp
RigidBodyInertia PluckerTransform::tformRBI(const RigidBodyInertia &Ihat) const
{
    auto m = Ihat.getMass();
    auto h = Ihat.getCom();
    auto I = Ihat.getInertiaMatrixLT();

    auto y = h - m * translation;
    auto h_new = rotation * y;

    auto Z = I + LowerTriangular::fromFullMatrix(skew(translation) * skew(h) + skew(y) * skew(translation));
    lt I_new = LowerTriangular::fromFullMatrix(rotation * Z * rotation.transpose());

    return RigidBodyInertia(m, h_new, I_new);
}
```

```cpp
RigidBodyInertia PluckerTransform::invtformRBI(const RigidBodyInertia &Ihat) const
{
    auto m = Ihat.getMass();
    auto h = Ihat.getCom();
    auto I = Ihat.getInertiaMatrixLT();

    auto h_new = rotation.transpose() * h + m * translation;
    auto I1 = rotation.transpose() * I * rotation;
    auto I2 = skew(translation) * skew(rotation.transpose() * h);
    auto I3 = skew(h_new) * skew(translation);

    lt I_new = LowerTriangular::fromFullMatrix(I1 - I2 - I3);

    return RigidBodyInertia(m, h_new, I_new);
}
```

```cpp
PluckerTransform PluckerTransform::inverse() const
{
    Rotation invRotation = rotation.transpose();
    Vector3d invTranslation = -invRotation * translation;
    return PluckerTransform(invRotation, invTranslation);
}
```

    Review tformRBI() implementation against Featherstone:
    
    **Featherstone (2008) Eq 2.52:**
    For rigid body inertia transform I' = X*I*X^T:
    - m' = m
    - h' = R*(h - m*r)
    - I' = R*(I + (r̂*ĥ + ĥ*r̂)) * R^T
      where ĥ = h - m*r
    
    **Current implementation (lines 65-78):**
    ```cpp
    auto y = h - m * translation;  // y = h - m*r ✓
    auto h_new = rotation * y;     // h' = R*y ✓
    auto Z = I + LowerTriangular::fromFullMatrix(
        skew(translation) * skew(h) + skew(y) * skew(translation)
    );
    lt I_new = LowerTriangular::fromFullMatrix(rotation * Z * rotation.transpose());
    ```
    
    **Verification:**
    - The formula uses skew(translation)*skew(h) + skew(y)*skew(translation)
    - This should represent: r̂*ĥ + ĥ*r̂ where ĥ = y
    - But it uses skew(h) not skew(y) in the first term
    - **POTENTIAL BUG:** Should be skew(translation)*skew(y) + skew(y)*skew(translation)
    
    **Action:**
    1. Cross-reference with Featherstone Eq 2.52
