Helper 1 — build FD solver from RobotModel:
```cpp
static SpatialAlgebra::ForwardDynamics buildFD(const RobotModel& model, int dof) {
    SpatialAlgebra::ForwardDynamics fd;
    fd.links.reserve(dof);
    for (int i = 0; i < dof; ++i) {
        const auto& js = model.joints[i];
        SpatialAlgebra::Link link;
        link.parent = js.parent;
        // Convert Matrix4d → PluckerTransform per Pitfall 3:
        link.X = SpatialAlgebra::PluckerTransform(
            SpatialAlgebra::Rotation(js.parentToJoint.topLeftCorner<3,3>()),
            js.parentToJoint.topRightCorner<3,1>()
        );
        // Convert mass+com+dense-inertia → RigidBodyInertia per Pitfall 2 (symmetrize):
        Eigen::Matrix3d symInertia = 0.5 * (js.inertia + js.inertia.transpose());
        link.I = SpatialAlgebra::RigidBodyInertia(
            js.mass, js.com,
            SpatialAlgebra::LowerTriangular::fromFullMatrix(symInertia)
        );
        // Convert Vector3d + JointType → MotionVector (screw axis):
        switch (js.type) {
            case JointType::REVOLUTE:
                link.S = SpatialAlgebra::MotionVector(js.jointAxis, SpatialAlgebra::Vector3d::Zero());
                break;
            case JointType::PRISMATIC:
                link.S = SpatialAlgebra::MotionVector(SpatialAlgebra::Vector3d::Zero(), js.jointAxis);
                break;
            case JointType::FIXED:
                link.S = SpatialAlgebra::MotionVector(SpatialAlgebra::Vector3d::Zero(), SpatialAlgebra::Vector3d::Zero());
                break;
        }
        // State initialized to zero:
        link.q = 0.0;
        link.qdot = 0.0;
        link.f = SpatialAlgebra::ForceVector(SpatialAlgebra::Vector3d::Zero(), SpatialAlgebra::Vector3d::Zero());
        fd.links.push_back(std::move(link));
    }
    return fd;
}
```
