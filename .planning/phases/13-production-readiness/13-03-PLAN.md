    **Part A — Create `include/spatial_assert.h`:**
    ```cpp
    #ifndef SPATIAL_ASSERT_H
    #define SPATIAL_ASSERT_H

    #include <iostream>
    #include <cstdlib>

    #ifndef NDEBUG

    /**
     * @brief Debug-only assertion macro for spatial algebra operations.
     * @details Aborts with a diagnostic message when the condition fails.
     *          Automatically stripped in release builds (NDEBUG defined).
     *          Use SPATIAL_ASSERT for invariant checks, SPATIAL_ASSERT_MSG
     *          when a custom message improves debugging clarity.
     */
    #define SPATIAL_ASSERT(cond)                                                \
        do {                                                                    \
            if (!(cond)) {                                                      \
                std::cerr << "SPATIAL_ASSERT: " #cond " failed at "             \
                          << __FILE__ << ":" << __LINE__ << std::endl;          \
                std::abort();                                                   \
            }                                                                   \
        } while (0)

    /**
     * @brief Debug-only assertion with custom error message.
     */
    #define SPATIAL_ASSERT_MSG(cond, msg)                                       \
        do {                                                                    \
            if (!(cond)) {                                                      \
                std::cerr << "SPATIAL_ASSERT: " #cond " failed at "             \
                          << __FILE__ << ":" << __LINE__ << " — "               \
                          << msg << std::endl;                                  \
                std::abort();                                                   \
            }                                                                   \
        } while (0)

    #else

    #define SPATIAL_ASSERT(cond)      ((void)0)
    #define SPATIAL_ASSERT_MSG(cond, msg) ((void)0)

    #endif // NDEBUG

    #endif // SPATIAL_ASSERT_H
    ```

    The `do { ... } while (0)` wrapper ensures the macros work correctly in all control-flow contexts (bare if/else branches). The `__FILE__` and `__LINE__` provide debuggable origin information. Use `std::abort()` (from `<cstdlib>`) for consistent non-ignorable failure — superior to raw `assert()` which can be intercepted by signal handlers and to bare `throw` which interacts poorly with debugger catchpoints.

    /**
     * @brief Test cross-force with mixed torque+force inputs
     * @details Exercises all 3 terms of the force×force formula:
     *          angular = τ1×τ2 + f1×f2, linear = τ1×f2 - τ2×f1
     */
    TEST(TestCrossProductForce, MixedTorqueForceInputs) {
        ForceVector a(Vector3d(1, 0, 0), Vector3d(0, 1, 0));
        ForceVector b(Vector3d(0, 1, 0), Vector3d(0, 0, 1));
        ForceVector result = cross(a, b);
        // angular: τ1×τ2 + f1×f2 = (0,0,1) + (1,0,0) = (1,0,1)
        EXPECT_NEAR(result.getAngular()[0], 1.0, EPSILON);
        EXPECT_NEAR(result.getAngular()[1], 0.0, EPSILON);
        EXPECT_NEAR(result.getAngular()[2], 1.0, EPSILON);
        // linear: τ1×f2 - τ2×f1 = (0,-1,0) - (0,0,0) = (0,-1,0)
        EXPECT_NEAR(result.getLinear()[0], 0.0, EPSILON);
        EXPECT_NEAR(result.getLinear()[1], -1.0, EPSILON);
        EXPECT_NEAR(result.getLinear()[2], 0.0, EPSILON);
    }

    TEST(TestCrossProductForce, AntiCommutativityMixedInputs) {
        ForceVector a(Vector3d(1, 2, 3), Vector3d(4, 5, 6));
        ForceVector b(Vector3d(0.5, -0.3, 1.2), Vector3d(-0.1, 0.8, -0.5));
        ForceVector ab = cross(a, b);
        ForceVector ba = cross(b, a);
        ForceVector neg_ba = ba * -1.0;
        EXPECT_NEAR((ab.getAngular() - neg_ba.getAngular()).norm(), 0.0, EPSILON);
        EXPECT_NEAR((ab.getLinear() - neg_ba.getLinear()).norm(), 0.0, EPSILON);
    }
