    /**
     * @brief RNEA static gravity torque is proportional to g
     * @details X-axis revolute joint, COM=(0,1,0), mass=1.0.
     *          Independent oracle: τ = r × m·g (Newton's law for point mass).
     *          For gravity = (0,0,g): τ[0] = m · g · 1.0 (lever arm = 1.0 along Y, gravity along Z)
     *          Invariant: τ[0] / g is constant across gravity levels.
     * @see MATHEMATICAL_CONVENTIONS.md Worked Example 1
     */
    TEST(StaticGravityInvariants, SingleLinkGravityProportionality) {
        double expected_ratio = -1.0;  // set on first iteration
        bool ratio_initialized = false;
        
        for (double g : {0.0, -4.905, -9.81}) {
            InverseDynamics id;
            InverseDynamicsLink link;
            link.parent = -1;
            link.X = PluckerTransform(Rotation(Eigen::Matrix3d::Identity()), Vector3d::Zero());
            link.I = RigidBodyInertia(1.0, Vector3d(0, 1.0, 0), lt::Identity(3));
            link.S = MotionVector(Vector3d(1, 0, 0), Vector3d::Zero());  // X-axis revolute
            link.q = 0.0; link.qdot = 0.0;
            id.links.push_back(link);
            
            Eigen::VectorXd qddot(1);
            qddot[0] = 0.0;  // zero acceleration — pure gravity compensation
            Eigen::VectorXd tau = id.computeTorques(qddot, Vector3d(0, 0, g));
            
            EXPECT_TRUE(std::isfinite(tau[0]));
            
            if (std::abs(g) > 1e-12) {
                double ratio = tau[0] / g;
                if (!ratio_initialized) {
                    expected_ratio = ratio;
                    ratio_initialized = true;
                }
                // τ/g should be constant (structural invariant)
                EXPECT_NEAR(ratio, expected_ratio, 1e-10)
                    << "Gravity proportionality invariant failed at g=" << g;
            } else {
                // Zero gravity → zero torque
                EXPECT_NEAR(tau[0], 0.0, 1e-10);
            }
        }
    }
