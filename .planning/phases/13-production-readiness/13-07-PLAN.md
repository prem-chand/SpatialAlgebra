    ```cpp
    /**
     * @brief Cross product of two zero ForceVectors returns zero
     * @details Invariant: cross(zero, zero) = zero (linearity of cross product).
     *          This is the minimum edge case: zero input → zero output.
     */
    TEST(EdgeCases, CrossForceZeroVectors) {
        ForceVector zero = ForceVector(Vector3d::Zero(), Vector3d::Zero());
        ForceVector result = cross(zero, zero);
        EXPECT_NEAR(result.getAngular().norm(), 0.0, 1e-15);
        EXPECT_NEAR(result.getLinear().norm(), 0.0, 1e-15);
    }
    ```

    ```cpp
    /**
     * @brief Cross product with zero linear component (pure torque)
     * @details If ForceVector a = (τ₁, 0), ForceVector b = (τ₂, 0):
     *          angular = τ₁×τ₂ + 0×0 = τ₁×τ₂
     *          linear  = τ₁×0 - τ₂×0 = 0
     *          Both angular and linear should produce correct results.
     */
    TEST(EdgeCases, CrossForceZeroLinear) {
        ForceVector a(Vector3d(1.0, 0.0, 0.0), Vector3d::Zero());
        ForceVector b(Vector3d(0.0, 1.0, 0.0), Vector3d::Zero());
        ForceVector result = cross(a, b);
        // angular = τ₁×τ₂ = (1,0,0)×(0,1,0) = (0,0,1)
        EXPECT_NEAR(result.getAngular()[0], 0.0, 1e-15);
        EXPECT_NEAR(result.getAngular()[1], 0.0, 1e-15);
        EXPECT_NEAR(result.getAngular()[2], 1.0, 1e-15);
        // linear = 0 (both inputs have zero linear component)
        EXPECT_NEAR(result.getLinear().norm(), 0.0, 1e-15);
    }
    ```

    ```cpp
    /**
     * @brief Zero-mass ABA edge case: contract says REJECTED
     * @details Zero mass produces degenerate joint-space inertia in the ABA
     *          inward pass (denominator near zero). The solver throws
     *          std::runtime_error. This is the CONTRACT: callers must ensure
     *          mass > 0.
     * @see MATHEMATICAL_CONVENTIONS.md — Edge Case Policy: Zero Mass
     */
    TEST(ForwardDynamicsTest, EdgeCase_ZeroMassThrows) {
        ForwardDynamics fd;
        Link link;
        link.parent = -1;
        link.X = PluckerTransform(Rotation(Eigen::Matrix3d::Identity()), Vector3d::Zero());
        link.I = RigidBodyInertia(0.0, Vector3d::Zero(), lt::Identity(3));
        link.S = MotionVector(Vector3d(0, 0, 1), Vector3d::Zero());
        link.q = 0.0; link.qdot = 0.0;
        link.f = ForceVector(Vector3d::Zero(), Vector3d::Zero());
        fd.links.push_back(link);
        
        Eigen::VectorXd tau(1);
        tau[0] = 1.0;
        // Zero mass should throw (degenerate inertia) — this IS the contract
        EXPECT_THROW(fd.computeAccelerations(tau), std::runtime_error);
    }
    ```

    ```cpp
    /**
     * @brief Release-mode stability for ABA
     * @details Validates that ABA produces finite, physically-sensible output
     *          for valid inputs regardless of build mode (NDEBUG defined or not).
     *          Invariant checks: 1) All outputs finite, 2) Positive torque → positive acceleration,
     *          3) Gravity changes acceleration magnitude, 4) Multi-link: base ≠ tip acceleration.
     * @note This test uses ONLY valid (non-degenerate) inputs that work in both debug and release.
     */
    TEST(ForwardDynamicsTest, ReleaseModeStability) {
        // Single-link test
        ForwardDynamics fd;
        Link l0;
        l0.parent = -1;
        l0.X = PluckerTransform(Rotation(Eigen::Matrix3d::Identity()), Vector3d::Zero());
        l0.I = RigidBodyInertia(1.0, Vector3d(0, 1.0, 0), lt::Identity(3));
        l0.S = MotionVector(Vector3d(1, 0, 0), Vector3d::Zero());
        l0.q = 0.0; l0.qdot = 0.0;
        l0.f = ForceVector(Vector3d::Zero(), Vector3d::Zero());
        fd.links.push_back(l0);
        
        Eigen::VectorXd tau(1);
        tau[0] = 1.0;
        
        // Invariant 1: No gravity → positive acceleration (torque dominates)
        fd.computeAccelerations(tau, Vector3d::Zero());
        EXPECT_TRUE(std::isfinite(fd.links[0].qddot));
        EXPECT_GT(fd.links[0].qddot, 0.0);
        double qddot_no_gravity = fd.links[0].qddot;
        
        // Invariant 2: With gravity → acceleration differs (gravity not zero)
        fd.computeAccelerations(tau, Vector3d(0, 0, -9.81));
        EXPECT_TRUE(std::isfinite(fd.links[0].qddot));
        EXPECT_GT(fd.links[0].qddot, 0.0);
        // With COM offset, gravity should change acceleration (no exact relationship, just different)
        // (This is the invariant: gravity != zero → different dynamics)
        bool gravityAffects = (std::abs(fd.links[0].qddot - qddot_no_gravity) > 1e-12);
        EXPECT_TRUE(gravityAffects) << "Gravity should affect ABA acceleration";
        
        // Invariant 3: Two-link with gravity → all finite, base accelerates less than tip
        Link l1;
        l1.parent = 0;
        l1.X = PluckerTransform(Rotation(Eigen::Matrix3d::Identity()), Vector3d(1, 0, 0));
        l1.I = RigidBodyInertia(1.0, Vector3d(0, 1.0, 0), lt::Identity(3));
        l1.S = MotionVector(Vector3d(1, 0, 0), Vector3d::Zero());
        l1.q = 0.0; l1.qdot = 0.0;
        l1.f = ForceVector(Vector3d::Zero(), Vector3d::Zero());
        fd.links.push_back(l1);
        
        Eigen::VectorXd tau2(2);
        tau2 << 1.0, 0.5;
        fd.computeAccelerations(tau2, Vector3d(0, 0, -9.81));
        
        EXPECT_TRUE(std::isfinite(fd.links[0].qddot));
        EXPECT_TRUE(std::isfinite(fd.links[1].qddot));
        // Structural invariant: two distinct links → two distinct accelerations (CR-02 permitting)
        EXPECT_NE(fd.links[0].qddot, fd.links[1].qddot);
    }
    ```
